<?php

function slugify(string $title): string
{
    $slug = strtolower(trim($title));
    $slug = preg_replace('/[^a-z0-9]+/', '-', $slug);
    return trim($slug, '-');
}

echo slugify("Hello World, again!"); // hello-world-again

<?php

final class Temperature
{
    public function __construct(private float $celsius)
    {
    }

    public function toFahrenheit(): float
    {
        return $this->celsius * 9 / 5 + 32;
    }
}

$t = new Temperature(21.5);
echo $t->toFahrenheit();

<?php

function findUser(PDO $db, string $email): ?array
{
    $stmt = $db->prepare('SELECT id, name FROM users WHERE email = :email');
    $stmt->execute([':email' => $email]);
    $row = $stmt->fetch(PDO::FETCH_ASSOC);
    return $row === false ? null : $row;
}

function validateForm(array $input): array
{
    $errors = [];
    if (empty($input['name'])) {
        $errors['name'] = 'Name is required';
    }
    if (!filter_var($input['email'] ?? '', FILTER_VALIDATE_EMAIL)) {
        $errors['email'] = 'Email is not valid';
    }
    return $errors;
}

interface Notifier
{
    public function send(string $to, string $message): bool;
}

class MailNotifier implements Notifier
{
    public function send(string $to, string $message): bool
    {
        return mail($to, 'Notice', $message);
    }
}

<?php

enum Suit: string
{
    case Hearts = 'H';
    case Spades = 'S';

    public function color(): string
    {
        return match ($this) {
            Suit::Hearts => 'Red',
            Suit::Spades => 'Black',
        };
    }
}

<?php

function readCsv(string $path): array
{
    $rows = [];
    $handle = fopen($path, 'r');
    if ($handle === false) {
        throw new RuntimeException("Cannot open $path");
    }
    while (($line = fgetcsv($handle)) !== false) {
        $rows[] = $line;
    }
    fclose($handle);
    return $rows;
}
